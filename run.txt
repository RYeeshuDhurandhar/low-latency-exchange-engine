Run core logic:

mkdir -p build
cmake -S . -B build
cmake --build build
./build/exchange_engine

--------------------------------------------------------------------------------------------------------
Run benchmark v1:

cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build

Run one workload:
./build/exchange_benchmark --impl map --workload mostly_adds --orders 10000000

Run all workloads:
./build/exchange_benchmark --impl map --all-workloads --orders 10000000

Write CSV:
./build/exchange_benchmark --impl map --all-workloads --orders 10000000 --csv results/results_map.csv

Use fixed seed:
./build/exchange_benchmark --impl map --all-workloads --orders 10000000 --seed 1 --csv results/results_map.csv

After adding ladder:

./build/exchange_benchmark --impl ladder --all-workloads --orders 10000000 --seed 1 --csv results/results_ladder.csv

--------------------------------------------------------------------------------------------------------
Run benchmark v2:

cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build

./build/exchange_benchmark \
  --impl map \
  --all-workloads \
  --orders 10000000 \
  --seed 1 \
  --csv results/results_map_v3_compare.csv


./build/exchange_benchmark \
  --impl ladder_pool \
  --all-workloads \

  --seed 1 \
  --csv results/results_ladder_pool_v3.csv

Debug mode:

cmake -S . -B build-debug \

  -DCMAKE_EXE_LINKER_FLAGS_DEBUG="-fsanitize=address,undefined"

cmake --build build-debug

./build-debug/exchange_engine

Run core logic:

mkdir -p build
cmake -S . -B build
cmake --build build
./build/exchange_engine

--------------------------------------------------------------------------------------------------------
Run benchmark v1:

cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build

Run one workload:
./build/exchange_benchmark --book_type map --workload mostly_adds --orders 10000000

Run all workloads:
./build/exchange_benchmark --book_type map --all-workloads --orders 10000000

Write CSV:
./build/exchange_benchmark --book_type map --all-workloads --orders 10000000 --csv results/results_map.csv

Use fixed seed:
./build/exchange_benchmark --book_type map --all-workloads --orders 10000000 --seed 1 --csv results/results_map.csv

After adding ladder:

./build/exchange_benchmark --book_type ladder --all-workloads --orders 10000000 --seed 1 --csv results/results_ladder.csv

Debug mode:

  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_CXX_FLAGS_DEBUG="-O0 -g -fsanitize=address,undefined -fno-omit-frame-pointer" \
  -DCMAKE_EXE_LINKER_FLAGS_DEBUG="-fsanitize=address,undefined"

cmake --build build-debug

./build-debug/exchange_engine
